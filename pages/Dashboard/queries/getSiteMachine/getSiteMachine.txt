SELECT 
  s.site_name AS "Site Name",
  m.id,
  m.machine_name AS "PC Name",
  m.machine_notes AS "Notes",
	m.created_at AS "Create Date",
	m.software_version AS "Software Version",
	m.last_backup AS "Last Backup",
	m.activation_code AS "Activation Code",
	m.msp_id,
	m.site_id,
	m.install_id AS "Machine Key",
  mt.name AS "Type",
	ms.latest_version AS "Latest Version",
  msu.name AS "Software Status"
FROM machines m
JOIN msps ms ON m.msp_id = ms.id
JOIN sites s ON m.site_id = s.id AND ms.id = m.msp_id
LEFT JOIN machine_type_lookup mt ON m.machine_type = mt.value
LEFT JOIN machine_status_lookup msu ON m.machine_status = msu.value
WHERE m.msp_id = {{appsmith.store.currentMsp}}
 AND m.id = {{this.params.machineId}}
 ORDER BY "Last Backup" DESC;
